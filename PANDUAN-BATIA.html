🛠️ TROUBLESHOOTING

1. ERR_PACKAGE_PATH_NOT_EXPORTED ./v3 (lepas pnpm install semula)

cd D:\batia-os
$p = "node_modules\.pnpm\tiktok-live-connector@2.4.3_debug@4.4.3\node_modules\tiktok-live-proto\package.json"
$c = Get-Content $p -Raw
$pattern = '"import": \{\s*"types": "(\./dist/node/(v\d)\.d\.ts)",\s*"default": "\./dist/node/\2\.js"\s*\}'
$repl = '$0,' + "`n" + '      "require": { "types": "./dist/node/$2.d.ts", "default": "./dist/node/$2.js" }'
$c2 = $c -replace $pattern, $repl
Set-Content -Path $p -Value $c2 -Encoding ascii

2. Emoji jadi ??
